# Raspberry Pi boot configuration for the universal raspir-blaster SD card.
# Copied as-is to build/sdcard/config.txt. Selects the kernel image and the
# CPU mode per board (see openspec/specs/board-support).

initial_turbo=0

# Raspberry Pi Zero W (BCM2835, 32 bit)
[pi0w]
arm_64bit=0
kernel=kernel.img

# Raspberry Pi Zero 2 W (64 bit)
[pi02]
arm_64bit=1
kernel_address=0x80000
kernel=kernel8.img

# Raspberry Pi 3 B, 3 A+ (64 bit)
[pi3]
arm_64bit=1
kernel_address=0x80000
kernel=kernel8.img

# Raspberry Pi 3 B+ (64 bit)
[pi3+]
arm_64bit=1
kernel_address=0x80000
kernel=kernel8.img

# Raspberry Pi 4 B and Compute Module 4 Lite (64 bit)
[pi4]
arm_64bit=1
kernel_address=0x80000
armstub=armstub8-rpi4.bin
kernel=kernel8-rpi4.img
max_framebuffers=2

[cm4]
otg_mode=1

[all]
